## アーキテクチャ方針

### 基本的な役割分担

**FastAPI(バックエンド)側で実装:**
- パスワード認証のロジック
- パスワードのハッシュ化
- JWTトークンの発行・検証
- データベース操作
- ビジネスロジック

**React(フロントエンド)側で実装（将来的に）:**
- ログイン/登録フォームのUI
- トークンの保存(localStorageなど)
- 認証が必要なリクエストへのトークン付与
- ユーザー体験(UX)

### 認証フロー
1. ユーザーがログインフォームに**メールアドレスとパスワード**を入力
2. React → FastAPI `/api/v1/auth/login` にPOST
3. FastAPI側でメールアドレスからアカウントを検索し、パスワードを検証
4. JWTトークンを生成して返却
5. Reactでトークンを`localStorage`に保存
6. 以降のAPIリクエストにトークンを付与（`Authorization: Bearer <token>`）

### 使用する`Account`モデルフィールド
- **`email`**: ログインIDとして使用
- **`hashed_password`**: パスワードのハッシュ値を保存
- **`display_name`**: ユーザー表示名
- **`is_active`**: アカウントの有効/無効を管理
- **`is_superuser`**: 管理者権限の管理（将来の拡張用）
- **`last_login`**: 最終ログイン時刻の記録

**今回は使用しないフィールド:**
- **`provider`**: 外部認証プロバイダー → NULL or "local"を設定
- **`external_id`**: 外部認証のユーザーID → NULL

```python
from typing import Optional
from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession
from app.models.accounts import Account
from app.repositories.base import BaseRepository
from datetime import datetime


class AccountRepository(BaseRepository[Account]):
    """アカウント管理用リポジトリ"""

    def __init__(self, db: AsyncSession):
        super().__init__(Account, db)

    async def get_by_email(self, email: str) -> Optional[Account]:
        """メールアドレスでアカウントを取得"""
        stmt = select(Account).where(Account.email == email)
        result = await self.db.execute(stmt)
        return result.scalar_one_or_none()

    async def create_account(
        self,
        email: str,
        hashed_password: str,
        display_name: str,
    ) -> Account:
        """新規アカウントを作成"""
        account = Account(
            email=email,
            hashed_password=hashed_password,
            display_name=display_name,
            provider="local",
            is_active=True,
            is_superuser=False,
        )
        self.db.add(account)
        await self.db.commit()
        await self.db.refresh(account)
        return account

    async def update_last_login(self, account_id: int) -> None:
        """最終ログイン時刻を更新"""
        stmt = (
            update(Account)
            .where(Account.id == account_id)
            .values(last_login=datetime.utcnow())
        )
        await self.db.execute(stmt)
        await self.db.commit()

    async def update_password(
        self, account_id: int, hashed_password: str
    ) -> None:
        """パスワードを更新"""
        stmt = (
            update(Account)
            .where(Account.id == account_id)
            .values(hashed_password=hashed_password)
        )
        await self.db.execute(stmt)
        await self.db.commit()

    async def update_display_name(
        self, account_id: int, display_name: str
    ) -> None:
        """表示名を更新"""
        stmt = (
            update(Account)
            .where(Account.id == account_id)
            .values(display_name=display_name)
        )
        await self.db.execute(stmt)
        await self.db.commit()

    async def deactivate_account(self, account_id: int) -> None:
        """アカウントを無効化"""
        stmt = (
            update(Account)
            .where(Account.id == account_id)
            .values(is_active=False)
        )
        await self.db.execute(stmt)
        await self.db.commit()
```

### 3. AuthServiceクラス実装

```python
from datetime import datetime, timedelta
from typing import Optional
from jose import JWTError, jwt
from passlib.context import CryptContext
from app.repositories.account_repository import AccountRepository
from app.models.accounts import Account
from app.utils.config import SECRET_KEY, ALGORITHM, ACCESS_TOKEN_EXPIRE_MINUTES


class AuthService:
    """認証サービス"""

    def __init__(self, account_repository: AccountRepository):
        self.account_repository = account_repository
        self.pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

    def hash_password(self, password: str) -> str:
        """パスワードをハッシュ化"""
        return self.pwd_context.hash(password)

    def verify_password(self, plain_password: str, hashed_password: str) -> bool:
        """パスワードを検証"""
        return self.pwd_context.verify(plain_password, hashed_password)

    def create_access_token(
        self, data: dict, expires_delta: Optional[timedelta] = None
    ) -> str:
        """JWTアクセストークンを生成"""
        to_encode = data.copy()
        if expires_delta:
            expire = datetime.utcnow() + expires_delta
        else:
            expire = datetime.utcnow() + timedelta(
                minutes=ACCESS_TOKEN_EXPIRE_MINUTES
            )
        to_encode.update({"exp": expire})
        encoded_jwt = jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)
        return encoded_jwt

    def decode_access_token(self, token: str) -> Optional[dict]:
        """JWTトークンをデコード"""
        try:
            payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
            return payload
        except JWTError:
            return None

    async def authenticate_user(
        self, email: str, password: str
    ) -> Optional[Account]:
        """ユーザー認証"""
        account = await self.account_repository.get_by_email(email)
        if not account:
            return None
        if not self.verify_password(password, account.hashed_password):
            return None
        if not account.is_active:
            return None
        return account

    async def register_user(
        self, email: str, password: str, display_name: str
    ) -> Account:
        """新規ユーザー登録"""
        hashed_password = self.hash_password(password)
        account = await self.account_repository.create_account(
            email=email,
            hashed_password=hashed_password,
            display_name=display_name,
        )
        return account
```

```python
from datetime import datetime
from typing import Optional
from pydantic import BaseModel, EmailStr, Field, validator


# ===== リクエストスキーマ =====

class AccountRegisterRequest(BaseModel):
    """アカウント登録リクエスト"""
    email: EmailStr = Field(..., description="メールアドレス")
    password: str = Field(
        ...,
        min_length=8,
        max_length=128,
        description="パスワード（8文字以上）",
    )
    display_name: str = Field(
        ...,
        min_length=1,
        max_length=100,
        description="表示名",
    )

    @validator("password")
    def validate_password_strength(cls, v: str) -> str:
        """パスワード強度の検証"""
        if not any(char.isdigit() for char in v):
            raise ValueError("パスワードには少なくとも1つの数字を含める必要があります")
        if not any(char.isalpha() for char in v):
            raise ValueError("パスワードには少なくとも1つの文字を含める必要があります")
        return v


class AccountLoginRequest(BaseModel):
    """アカウントログインリクエスト"""
    email: EmailStr = Field(..., description="メールアドレス")
    password: str = Field(..., description="パスワード")


class AccountUpdateRequest(BaseModel):
    """アカウント情報更新リクエスト"""
    display_name: Optional[str] = Field(
        None,
        min_length=1,
        max_length=100,
        description="表示名",
    )
    email: Optional[EmailStr] = Field(None, description="メールアドレス")


class PasswordChangeRequest(BaseModel):
    """パスワード変更リクエスト"""
    current_password: str = Field(..., description="現在のパスワード")
    new_password: str = Field(
        ...,
        min_length=8,
        max_length=128,
        description="新しいパスワード（8文字以上）",
    )

    @validator("new_password")
    def validate_password_strength(cls, v: str) -> str:
        """パスワード強度の検証"""
        if not any(char.isdigit() for char in v):
            raise ValueError("パスワードには少なくとも1つの数字を含める必要があります")
        if not any(char.isalpha() for char in v):
            raise ValueError("パスワードには少なくとも1つの文字を含める必要があります")
        return v


# ===== レスポンススキーマ =====

class AccountResponse(BaseModel):
    """アカウント情報レスポンス"""
    id: int
    email: str
    display_name: str
    is_active: bool
    is_superuser: bool
    last_login: Optional[datetime]
    created_at: datetime
    updated_at: datetime

    class Config:
        from_attributes = True


class TokenResponse(BaseModel):
    """トークンレスポンス"""
```

### 2. 認証エンドポイント実装

```python
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from app.api.dependencies.database import get_db
from app.repositories.account_repository import AccountRepository
from app.services.auth_service import AuthService
from app.schemas.accounts import (
    AccountRegisterRequest,
    AccountLoginRequest,
    AccountResponse,
    TokenResponse,
)
from app.exceptions.business import DuplicateEmailError
from app.exceptions.validation import InvalidCredentialsError


router = APIRouter(prefix="/auth", tags=["認証"])


@router.post(
    "/register",
    response_model=AccountResponse,
    status_code=status.HTTP_201_CREATED,
    summary="新規アカウント登録",
    description="メールアドレスとパスワードで新しいアカウントを登録します。",
)
async def register(
    request: AccountRegisterRequest,
    db: AsyncSession = Depends(get_db),
) -> AccountResponse:
    """新規アカウント登録"""
    account_repo = AccountRepository(db)
    auth_service = AuthService(account_repo)

    # メールアドレスの重複チェック
    existing_account = await account_repo.get_by_email(request.email)
    if existing_account:
        raise DuplicateEmailError(email=request.email)

    # 新規ユーザー登録
    account = await auth_service.register_user(
        email=request.email,
        password=request.password,
        display_name=request.display_name,
    )

    return AccountResponse.from_orm(account)


@router.post(
    "/login",
    response_model=TokenResponse,
    summary="ログイン",
    description="メールアドレスとパスワードでログインし、JWTトークンを発行します。",
)
async def login(
    request: AccountLoginRequest,
    db: AsyncSession = Depends(get_db),
) -> TokenResponse:
    """ログイン"""
    account_repo = AccountRepository(db)
    auth_service = AuthService(account_repo)

    # ユーザー認証
    account = await auth_service.authenticate_user(
        email=request.email,
        password=request.password,
    )
    if not account:
        raise InvalidCredentialsError()

    # 最終ログイン時刻を更新
    await account_repo.update_last_login(account.id)

    # JWTトークン生成
    access_token = auth_service.create_access_token(
        data={"sub": str(account.id), "email": account.email}
    )

    return TokenResponse(access_token=access_token, token_type="bearer")
```

```python
from fastapi import APIRouter, Depends, HTTPException, status
from sqlalchemy.ext.asyncio import AsyncSession
from app.api.dependencies.database import get_db
from app.api.dependencies.auth import get_current_active_user
from app.repositories.account_repository import AccountRepository
from app.services.auth_service import AuthService
from app.models.accounts import Account
from app.schemas.accounts import (
    AccountResponse,
    AccountUpdateRequest,
    PasswordChangeRequest,
)
from app.exceptions.validation import InvalidCredentialsError


router = APIRouter(prefix="/accounts", tags=["アカウント管理"])


@router.get(
    "/me",
    response_model=AccountResponse,
    summary="現在のアカウント情報取得",
    description="認証済みユーザーの自分のアカウント情報を取得します。",
)
async def get_my_account(
    current_user: Account = Depends(get_current_active_user),
) -> AccountResponse:
    """現在のアカウント情報取得"""
    return AccountResponse.from_orm(current_user)


@router.put(
    "/me",
    response_model=AccountResponse,
    summary="アカウント情報更新",
    description="認証済みユーザーの表示名やメールアドレスを更新します。",
)
async def update_my_account(
    request: AccountUpdateRequest,
    current_user: Account = Depends(get_current_active_user),
    db: AsyncSession = Depends(get_db),
) -> AccountResponse:
    """アカウント情報更新"""
    account_repo = AccountRepository(db)

    if request.display_name:
        await account_repo.update_display_name(
            current_user.id, request.display_name
        )

    if request.email and request.email != current_user.email:
        # メールアドレスの重複チェック
        existing_account = await account_repo.get_by_email(request.email)
        if existing_account:
            from app.exceptions.business import DuplicateEmailError
            raise DuplicateEmailError(email=request.email)

        # メールアドレス更新（実装は追加メソッドが必要）
        # await account_repo.update_email(current_user.id, request.email)
        raise HTTPException(
            status_code=status.HTTP_501_NOT_IMPLEMENTED,
            detail="メールアドレスの更新は未実装です",
        )

    # 更新後のアカウント情報を取得
    updated_account = await account_repo.get_by_id(current_user.id)
    return AccountResponse.from_orm(updated_account)


@router.put(
    "/me/password",
    status_code=status.HTTP_204_NO_CONTENT,
    summary="パスワード変更",
    description="認証済みユーザーのパスワードを変更します。",
)
async def change_password(
    request: PasswordChangeRequest,
    current_user: Account = Depends(get_current_active_user),
    db: AsyncSession = Depends(get_db),
) -> None:
    """パスワード変更"""
    account_repo = AccountRepository(db)
    auth_service = AuthService(account_repo)

    # 現在のパスワードを検証
    if not auth_service.verify_password(
        request.current_password, current_user.hashed_password
    ):
        raise InvalidCredentialsError()

    # 新しいパスワードをハッシュ化して更新
    new_hashed_password = auth_service.hash_password(request.new_password)
    await account_repo.update_password(current_user.id, new_hashed_password)
```

### 2. 認証依存性注入の実装

```python
from typing import Optional
from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.ext.asyncio import AsyncSession
from app.api.dependencies.database import get_db
from app.repositories.account_repository import AccountRepository
from app.services.auth_service import AuthService
from app.models.accounts import Account


security = HTTPBearer()


async def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: AsyncSession = Depends(get_db),
) -> Account:
    """
    JWTトークンから現在のユーザーを取得

    Args:
        credentials: HTTPベアラートークン
        db: データベースセッション

    Returns:
        Account: 認証済みアカウント

    Raises:
        HTTPException: トークンが無効な場合
    """
    token = credentials.credentials
    account_repo = AccountRepository(db)
    auth_service = AuthService(account_repo)

    # トークンをデコード
    payload = auth_service.decode_access_token(token)
    if payload is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="無効な認証トークンです",
            headers={"WWW-Authenticate": "Bearer"},
        )

    # ユーザーIDを取得
    user_id: Optional[str] = payload.get("sub")
    if user_id is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="トークンからユーザーIDを取得できませんでした",
            headers={"WWW-Authenticate": "Bearer"},
        )

    # ユーザーを取得
    account = await account_repo.get_by_id(int(user_id))
    if account is None:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="ユーザーが見つかりません",
            headers={"WWW-Authenticate": "Bearer"},
        )

    return account


async def get_current_active_user(
    current_user: Account = Depends(get_current_user),
) -> Account:
    """
    現在のアクティブユーザーを取得

    Args:
        current_user: 認証済みアカウント

    Returns:
        Account: アクティブなアカウント

    Raises:
        HTTPException: アカウントが無効な場合
    """
    if not current_user.is_active:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="アカウントが無効化されています",
        )
    return current_user


async def get_current_superuser(
    current_user: Account = Depends(get_current_active_user),
) -> Account:
    """
    現在のスーパーユーザーを取得（管理者権限チェック）

    Args:
        current_user: 認証済みアクティブアカウント

    Returns:
        Account: スーパーユーザーアカウント

    Raises:
        HTTPException: スーパーユーザーでない場合
    """
    if not current_user.is_superuser:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail="管理者権限が必要です",
        )
    return current_user
```
